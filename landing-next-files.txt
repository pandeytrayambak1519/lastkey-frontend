} from "react";

} from "react-router-dom";

import Logo from "../branding/Logo";

const navigationItems = [
  {
    label: "Features",
    path: ROUTES.FEATURES,
  },
  {
    label: "Security",
    path: ROUTES.SECURITY,
  },
  {
    label: "Pricing",
    path: ROUTES.PRICING,
  },
  {
    label: "About",
    path: ROUTES.ABOUT,
  },
];

export default function LandingNavbar() {
  const location = useLocation();

  const {
    isAuthenticated,
  } = useAuth();

  const [
    mobileMenuOpen,
    setMobileMenuOpen,
  ] = useState(false);

  const [
    resourcesOpen,
    setResourcesOpen,
  ] = useState(false);

  useEffect(() => {
    if (!mobileMenuOpen && !resourcesOpen) {
      return;
    }

    setMobileMenuOpen(false);
    setResourcesOpen(false);
  }, [location.pathname, mobileMenuOpen, resourcesOpen]);

  useEffect(() => {
    if (!mobileMenuOpen) {
      return undefined;
    }

    const previousOverflow =
      document.body.style.overflow;

    document.body.style.overflow =
      "hidden";

    function handleEscape(event) {
      if (event.key === "Escape") {
        setMobileMenuOpen(false);
      }
    }

    window.addEventListener(
      "keydown",
      handleEscape,
    );

    return () => {
      document.body.style.overflow =
        previousOverflow;

      window.removeEventListener(
        "keydown",
        handleEscape,
      );
    };
  }, [mobileMenuOpen]);

  return (
    <header className="sticky top-0 z-50 border-b border-slate-200/80 bg-white/90 backdrop-blur-xl">
      <div className="mx-auto flex h-18 max-w-[1440px] items-center px-4 sm:px-6 lg:px-8">
        <Logo />

        <nav className="ml-12 hidden items-center gap-1 lg:flex">
          {navigationItems.map(
            (item) => (
              <NavLink
                key={item.path}
                to={item.path}
                className={({
                  isActive,
                }) =>
                  [
                    "rounded-xl px-4 py-2.5 text-sm font-semibold transition",
                    isActive
                      ? "bg-blue-50 text-blue-700"
                      : "text-slate-600 hover:bg-slate-50 hover:text-slate-950",
                  ].join(" ")
                }
              >
                {item.label}
              </NavLink>
            ),
          )}

          <div className="relative">
            <button
              type="button"
              onClick={() =>
                setResourcesOpen(
                  (current) => !current,
                )
              }
              className="flex items-center gap-1 rounded-xl px-4 py-2.5 text-sm font-semibold text-slate-600 transition hover:bg-slate-50 hover:text-slate-950"
            >
              Resources

              <ChevronDown
                className={[
                  "h-4 w-4 transition-transform",
                  resourcesOpen
                    ? "rotate-180"
                    : "",
                ].join(" ")}
              />
            </button>

            {resourcesOpen && (
              <div className="absolute left-0 top-[calc(100%+10px)] w-72 rounded-2xl border border-slate-200 bg-white p-2 shadow-2xl shadow-slate-950/10">
                <Link
                  to={ROUTES.PRIVACY}
                  className="block rounded-xl p-3 transition hover:bg-slate-50"
                >
                  <p className="text-sm font-bold text-slate-900">
                    Privacy
                  </p>

                  <p className="mt-1 text-xs leading-5 text-slate-500">
                    Learn how your personal information is protected.
                  </p>
                </Link>

                <Link
                  to={ROUTES.TERMS}
                  className="block rounded-xl p-3 transition hover:bg-slate-50"
                >
                  <p className="text-sm font-bold text-slate-900">
                    Terms of service
                  </p>

                  <p className="mt-1 text-xs leading-5 text-slate-500">
                    Review the conditions for using LastKey.
                  </p>
                </Link>
              </div>
            )}
          </div>
        </nav>

        <div className="ml-auto hidden items-center gap-3 lg:flex">
          <div className="mr-2 flex items-center gap-2 text-xs font-semibold text-emerald-700">
            <ShieldCheck className="h-4 w-4" />

            Secure by design
          </div>

          {isAuthenticated ? (
            <Link to={ROUTES.DASHBOARD}>
              <Button
                size="medium"
                variant="dark"
              >
                Open dashboard
              </Button>
            </Link>
          ) : (
            <>
              <Link
                to={ROUTES.LOGIN}
                className="rounded-xl px-4 py-2.5 text-sm font-bold text-slate-600 transition hover:bg-slate-50 hover:text-slate-950"
              >
                Sign in
              </Link>

              <Link to={ROUTES.REGISTER}>
                <Button size="medium">
                  Create free account
                </Button>
              </Link>
            </>
          )}
        </div>

        <button
          type="button"
          onClick={() =>
            setMobileMenuOpen(true)
          }
          className="ml-auto flex h-10 w-10 items-center justify-center rounded-xl border border-slate-200 text-slate-700 transition hover:bg-slate-50 lg:hidden"
          aria-label="Open navigation menu"
        >
          <Menu className="h-5 w-5" />
        </button>
      </div>

      {mobileMenuOpen && (
        <div className="fixed inset-0 z-50 bg-white lg:hidden">
          <div className="flex h-18 items-center border-b border-slate-200 px-4 sm:px-6">
            <Logo />

            <button
              type="button"
              onClick={() =>
                setMobileMenuOpen(false)
              }
              className="ml-auto flex h-10 w-10 items-center justify-center rounded-xl border border-slate-200 text-slate-700"
              aria-label="Close navigation menu"
            >
              <X className="h-5 w-5" />
            </button>
          </div>

          <div className="flex h-[calc(100vh-72px)] flex-col overflow-y-auto px-4 py-6 sm:px-6">
            <nav className="space-y-2">
              {navigationItems.map(
                (item) => (
                  <NavLink
                    key={item.path}
                    to={item.path}
                    className={({
                      isActive,
                    }) =>
                      [
                        "block rounded-2xl px-4 py-4 text-base font-bold transition",
                        isActive
                          ? "bg-blue-50 text-blue-700"
                          : "text-slate-700 hover:bg-slate-50",
                      ].join(" ")
                    }
                  >
                    {item.label}
                  </NavLink>
                ),
              )}

              <Link
                to={ROUTES.PRIVACY}
                className="block rounded-2xl px-4 py-4 text-base font-bold text-slate-700 transition hover:bg-slate-50"
              >
                Privacy
              </Link>

              <Link
                to={ROUTES.TERMS}
                className="block rounded-2xl px-4 py-4 text-base font-bold text-slate-700 transition hover:bg-slate-50"
              >
                Terms
              </Link>
            </nav>

            <div className="mt-auto space-y-3 border-t border-slate-200 pt-6">
              {isAuthenticated ? (
                <Link
                  to={ROUTES.DASHBOARD}
                  className="block"
                >
                  <Button
                    fullWidth
                    variant="dark"
                  >
                    Open dashboard
                  </Button>
                </Link>
              ) : (
                <>
                  <Link
